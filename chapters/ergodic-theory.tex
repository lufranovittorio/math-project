\chapter{Ergodic Theory}\label{ergodic-theory}
\chapterstatus{planned}

Measure-preserving dynamics: recurrence, ergodic theorems, mixing and entropy.

\noindent\textbf{Prerequisites.} Chapter~\ref{measure-theory} (Measure and Integration), Chapter~\ref{functional-analysis} (Functional Analysis) and Section~\ref{ordinary-differential-equations:section-dynamical-systems} (Dynamical Systems).

\noindent\textbf{Planned contents.}
\begin{enumerate}
\item Measure-preserving transformations; Poincar\'e recurrence.
\item The von Neumann mean ergodic theorem and the Birkhoff pointwise ergodic theorem.
\item Ergodicity and mixing; examples: rotations, the doubling map, Bernoulli shifts, continued fractions.
\item Unique ergodicity and Weyl equidistribution.
\item Kolmogorov--Sinai entropy.
\item Exercises with solutions.
\end{enumerate}
