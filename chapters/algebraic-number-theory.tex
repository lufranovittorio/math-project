\chapter{Algebraic Number Theory}\label{algebraic-number-theory}
\chapterstatus{draft}

Many questions about the integers become transparent in a larger ring. Fermat's two-squares theorem is a statement about $\ZZ[i]$, the equation $y^2 = x^3 - 2$ about $\ZZ[\sqrt{-2}]$, and Fermat's
last theorem for the exponent $3$ about $\ZZ[\omega]$ (Chapters~\ref{number-theory} and~\ref{diophantine-equations}). In all these cases the ring has unique factorisation. Kummer and Dedekind found
how to proceed when it does not: the elements of a \emph{ring of integers} $\mathcal{O}_K$ of a number field need not factor uniquely, but its \emph{ideals} do. The failure of unique factorisation of
elements is measured by a finite abelian group, the \emph{class group}, and the units of $\mathcal{O}_K$ form a finitely generated group, by Dirichlet's unit theorem. Both finiteness theorems are proved
with the geometry of numbers. The chapter then studies how a rational prime splits in $\mathcal{O}_K$, the Frobenius elements of Galois extensions, and the cyclotomic fields, from which quadratic
reciprocity is rederived as a statement about the splitting of primes.

\noindent\textbf{Prerequisites.} Chapter~\ref{fields} (Fields and Galois Theory), Chapter~\ref{commutative-algebra} (Commutative Algebra), Chapter~\ref{number-theory} (Elementary Number Theory),
Chapter~\ref{geometry-of-numbers} (Geometry of Numbers) and Chapter~\ref{continued-fractions} (Continued Fractions and Diophantine Approximation).

\section{Number fields and their rings of integers}\label{algebraic-number-theory:section-number-fields}

\begin{definition}\label{algebraic-number-theory:definition-number-field}
A \emph{number field} is a finite extension $K$ of $\QQ$; its degree is $n = [K : \QQ]$. An \emph{algebraic integer} is a complex number which is a root of a monic polynomial with coefficients in
$\ZZ$. The \emph{ring of integers} $\mathcal{O}_K$ of $K$ is the set of algebraic integers in $K$, that is, the integral closure of $\ZZ$ in $K$
(Definition~\ref{commutative-algebra:definition-integral}).
\end{definition}

\begin{notation}\label{algebraic-number-theory:notation-number-fields}
Throughout, $K$ is a number field of degree $n$, regarded as a subfield of $\CC$ when convenient, and $\mathcal{O}_K$ its ring of integers. Ideals of $\mathcal{O}_K$ are written with gothic letters
$\mathfrak{a}, \mathfrak{b}, \mathfrak{p}$, and $\mathfrak{p}$ always denotes a nonzero prime ideal; $(\alpha) = \alpha\mathcal{O}_K$ is the principal ideal generated by $\alpha$. Products of ideals are written by
juxtaposition, $\mathfrak{a}\mathfrak{b}$. The embeddings (field homomorphisms) $K \to \CC$ are denoted $\sigma_1, \dots, \sigma_n$.
\end{notation}

By Theorem~\ref{fields:theorem-primitive-element}, $K = \QQ(\theta)$ for some $\theta$, and the $n$ embeddings send $\theta$ to the $n$ distinct roots in $\CC$ of its minimal polynomial; by
Theorem~\ref{fields:theorem-extension} every embedding of a subfield $F \subseteq K$ extends to exactly $[K : F]$ embeddings of $K$.

\begin{proposition}\label{algebraic-number-theory:proposition-algebraic-integers}
\begin{enumerate}
\item An algebraic number $\alpha$ is an algebraic integer if and only if its minimal polynomial over $\QQ$ has integer coefficients.
\item $\mathcal{O}_K$ is a subring of $K$, and $\mathcal{O}_K \cap \QQ = \ZZ$.
\item For every $\alpha \in K$ there is a nonzero $d \in \ZZ$ with $d\alpha \in \mathcal{O}_K$; hence $K$ is the fraction field of $\mathcal{O}_K$, and $K$ has a $\QQ$-basis contained in $\mathcal{O}_K$.
\item $\mathcal{O}_K$ is integrally closed in $K$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) If $f(\alpha) = 0$ with $f \in \ZZ[x]$ monic, the minimal polynomial $g$ divides $f$ in $\QQ[x]$, say $f = gh$ with $g$, $h$ monic. Let $a, b \geq 1$ be the least integers with $ag, bh \in \ZZ[x]$. Then
$ag$ is primitive (if a prime $p$ divided all its coefficients, $\frac ap g$ would have integer coefficients), and so is $bh$. By Gauss's lemma (Lemma~\ref{rings:lemma-gauss}) $abf = (ag)(bh)$ is
primitive; as $f$ is monic with integer coefficients, the content of $abf$ is $ab$, so $a = b = 1$ and $g \in \ZZ[x]$. The converse is clear.

(2) The integral closure is a subring by Proposition~\ref{commutative-algebra:proposition-integral}(2). A rational root of a monic integer polynomial is an integer (by (1), its minimal polynomial
$x - r$ is in $\ZZ[x]$).

(3) If $a_0\alpha^m + a_1\alpha^{m-1} + \dots + a_m = 0$ with $a_i \in \ZZ$ and $a_0 \neq 0$, multiplying by $a_0^{m-1}$ shows that $a_0\alpha$ is a root of the monic polynomial
$y^m + a_1y^{m-1} + a_0a_2y^{m-2} + \dots + a_0^{m-1}a_m$.

(4) An element of $K$ integral over $\mathcal{O}_K$ is integral over $\ZZ$, by transitivity (Proposition~\ref{commutative-algebra:proposition-integral}(2)).
\end{proof}

\subsection{Trace, norm and discriminant}\label{algebraic-number-theory:subsection-trace-norm}

\begin{definition}\label{algebraic-number-theory:definition-trace-norm}
For $\alpha \in K$, let $m_\alpha \colon K \to K$ be multiplication by $\alpha$, a $\QQ$-linear map. The \emph{trace} and \emph{norm} of $\alpha$ are
\[
\operatorname{Tr}_{K/\QQ}(\alpha) = \operatorname{tr}(m_\alpha), \qquad N_{K/\QQ}(\alpha) = \det(m_\alpha).
\]
We write $\operatorname{Tr}$ and $N$ when $K$ is clear.
\end{definition}

\begin{proposition}\label{algebraic-number-theory:proposition-trace-norm}
Let $\alpha \in K$ have minimal polynomial $f$ of degree $d$. Then the characteristic polynomial of $m_\alpha$ is $f^{n/d}$, and
\[
\operatorname{Tr}(\alpha) = \sum_{i=1}^n\sigma_i(\alpha), \qquad N(\alpha) = \prod_{i=1}^n\sigma_i(\alpha).
\]
In particular $\operatorname{Tr}$ is $\QQ$-linear, $N$ is multiplicative, and both take integer values on $\mathcal{O}_K$.
\end{proposition}

\begin{proof}
\emph{Step 1.} On $\QQ(\alpha)$, with the basis $1, \alpha, \dots, \alpha^{d-1}$, the matrix of $m_\alpha$ is the companion matrix of $f$, whose characteristic polynomial is $f$. If
$\beta_1, \dots, \beta_k$ is a basis of $K$ over $\QQ(\alpha)$, $k = n/d$, then $K = \bigoplus_j\QQ(\alpha)\beta_j$ is a decomposition into $m_\alpha$-stable subspaces, each isomorphic to $\QQ(\alpha)$, so the
characteristic polynomial of $m_\alpha$ on $K$ is $f^k$.

\emph{Step 2.} The roots of $f$ in $\CC$ are the images of $\alpha$ under the $d$ embeddings of $\QQ(\alpha)$, and each of these extends to exactly $k$ embeddings of $K$. So
$\prod_{i=1}^n(x - \sigma_i(\alpha)) = f^k$, and comparing coefficients of $x^{n-1}$ and constant terms gives the formulas.

\emph{Step 3.} If $\alpha \in \mathcal{O}_K$, then $f \in \ZZ[x]$ by Proposition~\ref{algebraic-number-theory:proposition-algebraic-integers}(1), so the coefficients of $f^k$, among them $\pm\operatorname{Tr}(\alpha)$ and $\pm N(\alpha)$, are integers.
\end{proof}

\begin{example}\label{algebraic-number-theory:example-trace-norm}
In $K = \QQ(\sqrt d)$, $d$ a squarefree integer, the embeddings send $\sqrt d$ to $\pm\sqrt d$, so $\operatorname{Tr}(a + b\sqrt d) = 2a$ and $N(a + b\sqrt d) = a^2 - db^2$. For $d = -1$ this is the norm
of $\ZZ[i]$ used in Chapter~\ref{number-theory}. An element $\alpha$ is a unit of $\mathcal{O}_K$ if and only if $N(\alpha) = \pm1$: if $\alpha\beta = 1$ then $N(\alpha)N(\beta) = 1$ with both integers, and
conversely $\alpha^{-1} = \pm\prod_{i \geq 2}\sigma_i(\alpha)$ is integral (in a Galois closure) and lies in $K$.
\end{example}

\begin{definition}\label{algebraic-number-theory:definition-discriminant}
The \emph{discriminant} of an $n$-tuple $(\alpha_1, \dots, \alpha_n)$ of elements of $K$ is
\[
d(\alpha_1, \dots, \alpha_n) = \det\bigl(\operatorname{Tr}(\alpha_i\alpha_j)\bigr)_{i,j} = \det\bigl(\sigma_i(\alpha_j)\bigr)_{i,j}^2.
\]
\end{definition}

The two expressions agree because $\operatorname{Tr}(\alpha_i\alpha_j) = \sum_k\sigma_k(\alpha_i)\sigma_k(\alpha_j)$, so the trace matrix is $S^TS$ with $S = (\sigma_k(\alpha_j))_{k,j}$. If
$(\beta_j) = (\alpha_j)C$ for a rational matrix $C$, then $d(\beta) = (\det C)^2d(\alpha)$. The \emph{sign} of a nonzero discriminant is $(-1)^{r_2}$, where $2r_2$ is the number of non-real embeddings:
complex conjugation permutes the rows of $S$, exchanging $r_2$ pairs, so $\overline{\det S} = (-1)^{r_2}\det S$, and $\det S$ is real or purely imaginary accordingly.

\begin{proposition}\label{algebraic-number-theory:proposition-discriminant-power-basis}
Let $K = \QQ(\theta)$, with $f$ the minimal polynomial of $\theta$ and $\theta_1, \dots, \theta_n$ its roots. Then
\[
d(1, \theta, \dots, \theta^{n-1}) = \prod_{i < j}(\theta_i - \theta_j)^2 = (-1)^{n(n-1)/2}N\bigl(f^{\prime}(\theta)\bigr) \neq 0.
\]
Consequently $d(\alpha_1, \dots, \alpha_n) \neq 0$ for every $\QQ$-basis, and the trace form $(x, y) \mapsto \operatorname{Tr}(xy)$ is nondegenerate.
\end{proposition}

\begin{proof}
$\det(\theta_i^{j})$ is a Vandermonde determinant, equal to $\pm\prod_{i < j}(\theta_i - \theta_j)$, which is nonzero since the roots are distinct. Next
$\prod_{i < j}(\theta_i - \theta_j)^2 = (-1)^{n(n-1)/2}\prod_{i \neq j}(\theta_i - \theta_j)$, and $\prod_{j \neq i}(\theta_i - \theta_j) = f^{\prime}(\theta_i)$, so the product is
$(-1)^{n(n-1)/2}\prod_if^{\prime}(\theta_i) = (-1)^{n(n-1)/2}N(f^{\prime}(\theta))$ by Proposition~\ref{algebraic-number-theory:proposition-trace-norm}. Any other basis differs by an invertible matrix.
\end{proof}

\begin{theorem}[Integral bases]\label{algebraic-number-theory:theorem-integral-basis}
$\mathcal{O}_K$ is a free $\ZZ$-module of rank $n$. A $\ZZ$-basis of $\mathcal{O}_K$ is called an \emph{integral basis}; all integral bases have the same discriminant $d_K$, the \emph{discriminant} of $K$,
which is a nonzero integer. If $\alpha_1, \dots, \alpha_n \in \mathcal{O}_K$ is a $\QQ$-basis of $K$ and $M = \bigoplus\ZZ\alpha_i$, then $d(\alpha_1, \dots, \alpha_n) = [\mathcal{O}_K : M]^2d_K$.
\end{theorem}

\begin{proof}
\emph{Step 1: $\mathcal{O}_K$ is sandwiched between lattices.} Choose a $\QQ$-basis $\alpha_1, \dots, \alpha_n$ of $K$ in $\mathcal{O}_K$
(Proposition~\ref{algebraic-number-theory:proposition-algebraic-integers}(3)), and let $\alpha_1^*, \dots, \alpha_n^*$ be the dual basis for the trace form, $\operatorname{Tr}(\alpha_i\alpha_j^*) = \delta_{ij}$,
which exists by Proposition~\ref{algebraic-number-theory:proposition-discriminant-power-basis}. If $x = \sum c_j\alpha_j^* \in \mathcal{O}_K$, then $c_i = \operatorname{Tr}(x\alpha_i) \in \ZZ$. So
$\bigoplus\ZZ\alpha_i \subseteq \mathcal{O}_K \subseteq \bigoplus\ZZ\alpha_i^*$.

\emph{Step 2: freeness.} A subgroup of a free abelian group of rank $n$ is free of rank at most $n$ (Theorem~\ref{geometry-of-numbers:theorem-discrete}, or the structure theorem for finitely generated abelian
groups), and $\mathcal{O}_K$ contains $\bigoplus\ZZ\alpha_i$ of rank $n$.

\emph{Step 3: the discriminant.} Two integral bases differ by a matrix in $\mathrm{GL}_n(\ZZ)$, of determinant $\pm1$, so they have the same discriminant; it is an integer by
Proposition~\ref{algebraic-number-theory:proposition-trace-norm} and nonzero by Proposition~\ref{algebraic-number-theory:proposition-discriminant-power-basis}. If $M = C\mathcal{O}_K$ in terms of an integral
basis, with $C$ an integer matrix, then $[\mathcal{O}_K : M] = |\det C|$ (Proposition~\ref{geometry-of-numbers:proposition-index}) and $d(\alpha) = (\det C)^2d_K$.
\end{proof}

\begin{theorem}[Quadratic fields]\label{algebraic-number-theory:theorem-quadratic-integers}
Let $d \neq 0, 1$ be a squarefree integer and $K = \QQ(\sqrt d)$. Then
\[
\mathcal{O}_K = \begin{cases} \ZZ[\sqrt d], & d_K = 4d, \quad \text{if } d \equiv 2, 3 \pmod 4,\\ \ZZ\bigl[\frac{1 + \sqrt d}{2}\bigr], & d_K = d, \quad \text{if } d \equiv 1 \pmod 4.\end{cases}
\]
\end{theorem}

\begin{proof}
\emph{Step 1: the condition.} $\alpha = a + b\sqrt d$ with $a, b \in \QQ$, $b \neq 0$, has minimal polynomial $x^2 - 2ax + (a^2 - db^2)$, so $\alpha \in \mathcal{O}_K$ if and only if $2a \in \ZZ$ and
$a^2 - db^2 \in \ZZ$. Write $a = A/2$, so $4db^2 = A^2 - 4(a^2 - db^2) \in \ZZ$; as $d$ is squarefree, $2b = B \in \ZZ$ (a prime in the denominator of $b$ would appear squared in that of $db^2$). The
condition becomes $A^2 - dB^2 \equiv 0 \pmod 4$.

\emph{Step 2: solving it.} Squares are $0$ or $1$ modulo $4$. If $d \equiv 2, 3 \pmod 4$, then $A^2 \equiv dB^2$ forces $A$, $B$ even, so $\mathcal{O}_K = \ZZ[\sqrt d]$. If $d \equiv 1 \pmod 4$, the
condition is $A \equiv B \pmod 2$, so $\mathcal{O}_K = \{\frac{A + B\sqrt d}{2} : A \equiv B \pmod 2\} = \ZZ[\frac{1 + \sqrt d}{2}]$.

\emph{Step 3: discriminants.} $d(1, \sqrt d) = \det\begin{pmatrix} 2 & 0\\ 0 & 2d\end{pmatrix} = 4d$, and $d(1, \frac{1 + \sqrt d}{2}) = \det\begin{pmatrix} 2 & 1\\ 1 & \frac{1 + d}{2}\end{pmatrix} = d$.
\end{proof}

\begin{example}\label{algebraic-number-theory:example-rings-of-integers}
$\mathcal{O}_{\QQ(i)} = \ZZ[i]$ with $d_K = -4$; $\mathcal{O}_{\QQ(\sqrt{-3})} = \ZZ[\omega]$ with $d_K = -3$ (since $\frac{1 + \sqrt{-3}}{2} = -\omega^2 = 1 + \omega$); $\mathcal{O}_{\QQ(\sqrt5)} = \ZZ[\varphi]$ with
$\varphi = \frac{1 + \sqrt5}{2}$ the golden ratio and $d_K = 5$; $\mathcal{O}_{\QQ(\sqrt{-5})} = \ZZ[\sqrt{-5}]$ with $d_K = -20$.
\end{example}

\begin{counterexample}\label{algebraic-number-theory:counterexample-not-maximal}
$\ZZ[\sqrt{-3}]$ is not the ring of integers of $\QQ(\sqrt{-3})$: it has index $2$ in $\ZZ[\omega]$, and it is not integrally closed, as $\omega$ is a root of $x^2 + x + 1$. In it, unique factorisation of
ideals fails: the ideal $\mathfrak{c} = (2, 1 + \sqrt{-3})$ satisfies $\mathfrak{c}^2 = 2\mathfrak{c}$ with $\mathfrak{c} \neq (2)$, so $\mathfrak{c}$ cannot be cancelled. (Indeed $(1 + \sqrt{-3})^2 = -2 + 2\sqrt{-3}$ and
$2(1 + \sqrt{-3})$, $4$ lie in $2\mathfrak{c}$, and conversely $2\mathfrak{c}$ is generated by $4$ and $2(1 + \sqrt{-3})$.) This is why one works with the full ring of integers.
\end{counterexample}

\begin{lemma}[Eisenstein polynomials]\label{algebraic-number-theory:lemma-eisenstein-index}
Let $\theta \in \mathcal{O}_K$ generate $K$, and suppose its minimal polynomial $f = x^n + c_{n-1}x^{n-1} + \dots + c_0$ is Eisenstein at a prime $p$: $p \mid c_i$ for all $i$ and $p^2 \nmid c_0$. Then
$p \nmid [\mathcal{O}_K : \ZZ[\theta]]$.
\end{lemma}

\begin{proof}
Suppose $p$ divides the index. Then the finite group $\mathcal{O}_K/\ZZ[\theta]$ has an element of order $p$: there is $x \in \mathcal{O}_K \setminus \ZZ[\theta]$ with
$px = \sum_{i=0}^{n-1}a_i\theta^i$, $a_i \in \ZZ$, not all divisible by $p$. Let $k$ be the least index with $p \nmid a_k$; subtracting $\sum_{i<k}\frac{a_i}{p}\theta^i \in \ZZ[\theta]$, we get
$y = \sum_{i \geq k}\frac{a_i}{p}\theta^i \in \mathcal{O}_K$. From $\theta^n = -\sum c_j\theta^j$ with $p \mid c_j$, every $\theta^m/p$ with $m \geq n$ lies in $\ZZ[\theta]$. Hence
$\theta^{n-1-k}y - \sum_{i > k}\frac{a_i}{p}\theta^{n-1-k+i} = \frac{a_k}{p}\theta^{n-1}$ lies in $\mathcal{O}_K$, and its norm $\frac{a_k^n}{p^n}N(\theta)^{n-1} = \pm\frac{a_k^nc_0^{n-1}}{p^n}$ is an integer. But
$p \nmid a_k$ and $p \,\|\, c_0$, so this number has $p$-adic valuation $(n - 1) - n = -1$, a contradiction.
\end{proof}

\section{Ideals in rings of integers}\label{algebraic-number-theory:section-ideals}

\begin{theorem}\label{algebraic-number-theory:theorem-dedekind}
$\mathcal{O}_K$ is a Dedekind domain (Definition~\ref{commutative-algebra:definition-regular}). Hence every nonzero ideal of $\mathcal{O}_K$ is uniquely a product of nonzero prime ideals.
\end{theorem}

\begin{proof}
\emph{Noetherian:} every ideal is a subgroup of $\mathcal{O}_K \cong \ZZ^n$, hence finitely generated. \emph{Normal:} Proposition~\ref{algebraic-number-theory:proposition-algebraic-integers}(4).
\emph{Dimension $1$:} if $\mathfrak{a} \neq 0$ is an ideal and $0 \neq \alpha \in \mathfrak{a}$, then $N(\alpha) = \alpha\prod_{i \geq 2}\sigma_i(\alpha)$ lies in $\mathfrak{a}$ (the cofactor
$N(\alpha)/\alpha$ is in $K$ and integral), so $\mathfrak{a}$ contains a nonzero integer $m$ and $\mathcal{O}_K/\mathfrak{a}$ is a quotient of $\mathcal{O}_K/m\mathcal{O}_K \cong (\ZZ/m)^n$, a finite ring. A
finite domain is a field, so nonzero primes are maximal. Unique factorisation is Theorem~\ref{commutative-algebra:theorem-dedekind}.
\end{proof}

\begin{definition}\label{algebraic-number-theory:definition-fractional-ideal}
A \emph{fractional ideal} of $\mathcal{O}_K$ is a nonzero finitely generated $\mathcal{O}_K$-submodule of $K$; equivalently, a set $\frac1d\mathfrak{a}$ with $\mathfrak{a}$ a nonzero ideal and
$d \in \ZZ \setminus \{0\}$. For a fractional ideal $\mathfrak{a}$ put $\mathfrak{a}^{-1} = \{x \in K : x\mathfrak{a} \subseteq \mathcal{O}_K\}$. The \emph{principal} fractional ideals are the $\alpha\mathcal{O}_K$,
$\alpha \in K^\times$.
\end{definition}

\begin{theorem}\label{algebraic-number-theory:theorem-fractional-ideals}
\begin{enumerate}
\item Every fractional ideal $\mathfrak{a}$ satisfies $\mathfrak{a}\mathfrak{a}^{-1} = \mathcal{O}_K$. The fractional ideals form an abelian group $I_K$ under multiplication, free on the nonzero prime ideals.
\item \emph{To contain is to divide}: for nonzero ideals, $\mathfrak{a} \subseteq \mathfrak{b}$ if and only if $\mathfrak{a} = \mathfrak{b}\mathfrak{c}$ for an ideal $\mathfrak{c}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) It suffices to treat a nonzero ideal $\mathfrak{a} = (a_1, \dots, a_r)$. The product $\mathfrak{j} = \mathfrak{a}\mathfrak{a}^{-1}$ is an ideal of $\mathcal{O}_K$. Suppose $\mathfrak{j} \subseteq \mathfrak{m}$ for a
maximal ideal $\mathfrak{m}$. The localisation $R = (\mathcal{O}_K)_\mathfrak{m}$ is a DVR (as in the proof of Theorem~\ref{commutative-algebra:theorem-dedekind}, via
Theorem~\ref{commutative-algebra:theorem-dvr}); let $v$ be its valuation and $a \in \mathfrak{a}$ an element with $v(a)$ minimal among the $v(a_i)$. Then $a_i/a \in R$, say $a_i/a = b_i/s_i$ with
$b_i \in \mathcal{O}_K$, $s_i \notin \mathfrak{m}$. Put $s = \prod s_i \notin \mathfrak{m}$ and $x = s/a$. Then $xa_i = sb_i/s_i \in \mathcal{O}_K$ for all $i$, so $x \in \mathfrak{a}^{-1}$ and
$s = xa \in \mathfrak{j} \subseteq \mathfrak{m}$, a contradiction. So $\mathfrak{j} = \mathcal{O}_K$. Inverses being available, the fractional ideals form a group; every fractional ideal is $\mathfrak{a}\mathfrak{b}^{-1}$
with $\mathfrak{a}$, $\mathfrak{b}$ ideals ($\frac1d\mathfrak{a} = \mathfrak{a}(d)^{-1}$), so by Theorem~\ref{algebraic-number-theory:theorem-dedekind} the group is generated by the primes, freely by uniqueness.

(2) If $\mathfrak{a} \subseteq \mathfrak{b}$, then $\mathfrak{c} = \mathfrak{b}^{-1}\mathfrak{a} \subseteq \mathfrak{b}^{-1}\mathfrak{b} = \mathcal{O}_K$ is an ideal with $\mathfrak{b}\mathfrak{c} = \mathfrak{a}$. The converse is clear.
\end{proof}

So for ideals, $\mathfrak{a} \subseteq \mathfrak{b}$ means that the exponents of $\mathfrak{a}$ are at least those of $\mathfrak{b}$; the sum $\mathfrak{a} + \mathfrak{b}$ is the greatest common divisor (minimum of exponents) and
$\mathfrak{a} \cap \mathfrak{b}$ the least common multiple.

\begin{definition}\label{algebraic-number-theory:definition-class-group}
The \emph{class group} of $K$ is $\mathrm{Cl}_K = I_K/P_K$, where $P_K$ is the subgroup of principal fractional ideals; its order $h_K$ is the \emph{class number}.
\end{definition}

$h_K = 1$ if and only if $\mathcal{O}_K$ is a PID, if and only if it is a UFD (a Dedekind domain which is a UFD is a PID: a prime ideal $\mathfrak{p}$ contains a prime factor $\pi$ of any of its nonzero
elements, and $(\pi) \subseteq \mathfrak{p}$ with $(\pi)$ a nonzero prime, hence maximal, so $\mathfrak{p} = (\pi)$; then every ideal is a product of principal ideals).

\begin{definition}\label{algebraic-number-theory:definition-ideal-norm}
The \emph{norm} of a nonzero ideal is $N(\mathfrak{a}) = |\mathcal{O}_K/\mathfrak{a}|$, finite by the proof of Theorem~\ref{algebraic-number-theory:theorem-dedekind}.
\end{definition}

\begin{proposition}\label{algebraic-number-theory:proposition-ideal-norm}
\begin{enumerate}
\item $N(\mathfrak{a}\mathfrak{b}) = N(\mathfrak{a})N(\mathfrak{b})$, so $N$ extends to a homomorphism $I_K \to \QQ_{>0}^\times$.
\item $N(\alpha\mathcal{O}_K) = |N_{K/\QQ}(\alpha)|$ for $\alpha \in K^\times$.
\item There are only finitely many ideals of a given norm, and $\mathfrak{a} \supseteq N(\mathfrak{a})\mathcal{O}_K$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) It suffices to show $N(\mathfrak{a}\mathfrak{p}) = N(\mathfrak{a})N(\mathfrak{p})$. As $|\mathcal{O}_K/\mathfrak{a}\mathfrak{p}| = |\mathcal{O}_K/\mathfrak{a}||\mathfrak{a}/\mathfrak{a}\mathfrak{p}|$, we show that $\mathfrak{a}/\mathfrak{a}\mathfrak{p}$
is a one-dimensional vector space over $\mathcal{O}_K/\mathfrak{p}$. An ideal $\mathfrak{j}$ with $\mathfrak{a}\mathfrak{p} \subseteq \mathfrak{j} \subseteq \mathfrak{a}$ is $\mathfrak{a}\mathfrak{c}$ with $\mathfrak{p} \subseteq \mathfrak{c} \subseteq \mathcal{O}_K$
(Theorem~\ref{algebraic-number-theory:theorem-fractional-ideals}), so $\mathfrak{c} \in \{\mathfrak{p}, \mathcal{O}_K\}$ and $\mathfrak{j} \in \{\mathfrak{a}\mathfrak{p}, \mathfrak{a}\}$, and these differ by cancellation. So
$\mathfrak{a}/\mathfrak{a}\mathfrak{p}$ is a nonzero module over the field $\mathcal{O}_K/\mathfrak{p}$ without proper nonzero submodules, that is, of dimension $1$.

(2) For $\alpha \in \mathcal{O}_K$: if $\omega_1, \dots, \omega_n$ is an integral basis, $\alpha\omega_1, \dots, \alpha\omega_n$ is a basis of $\alpha\mathcal{O}_K$, related to it by the matrix of $m_\alpha$, so the index is
$|\det m_\alpha|$ (Proposition~\ref{geometry-of-numbers:proposition-index}). The general case follows by multiplicativity.

(3) $\mathcal{O}_K/\mathfrak{a}$ is a group of order $N(\mathfrak{a})$, so $N(\mathfrak{a})$ kills it; there are finitely many ideals containing a given $m\mathcal{O}_K$, since $\mathcal{O}_K/m\mathcal{O}_K$ is finite.
\end{proof}

\begin{example}\label{algebraic-number-theory:example-sqrt-minus-five}
In $\ZZ[\sqrt{-5}]$ (Example~\ref{commutative-algebra:example-ideal-factorisation}), $\mathfrak{p} = (2, 1 + \sqrt{-5})$ has norm $2$ and is not principal, since $a^2 + 5b^2 = 2$ has no solution; similarly
$\mathfrak{r} = (3, 1 + \sqrt{-5})$ of norm $3$ is not principal. But $\mathfrak{p}^2 = (2)$ and $\mathfrak{p}\mathfrak{r} = (1 + \sqrt{-5})$, of norm $6$. So $\mathfrak{p}$ has order $2$ in $\mathrm{Cl}_K$, and
$\mathfrak{r}$ lies in the same class as $\mathfrak{p}^{-1} = \mathfrak{p}$.
\end{example}

\section{The decomposition of primes}\label{algebraic-number-theory:section-decomposition}

\begin{definition}\label{algebraic-number-theory:definition-ramification}
Let $p$ be a prime number and $p\mathcal{O}_K = \mathfrak{p}_1^{e_1}\cdots\mathfrak{p}_g^{e_g}$ with distinct primes $\mathfrak{p}_i$. The $\mathfrak{p}_i$ are the primes \emph{above} $p$ (equivalently, the primes with
$\mathfrak{p} \cap \ZZ = p\ZZ$); $e_i$ is the \emph{ramification index} and $f_i = [\mathcal{O}_K/\mathfrak{p}_i : \mathbb{F}_p]$ the \emph{residue degree} of $\mathfrak{p}_i$. The prime $p$ is \emph{ramified} if some
$e_i > 1$, \emph{inert} if $p\mathcal{O}_K$ is prime, and \emph{split completely} if $g = n$.
\end{definition}

\begin{theorem}\label{algebraic-number-theory:theorem-sum-ef}
$\sum_{i=1}^ge_if_i = n$.
\end{theorem}

\begin{proof}
By Proposition~\ref{algebraic-number-theory:proposition-ideal-norm}, $p^n = |N(p)| = N(p\mathcal{O}_K) = \prod_iN(\mathfrak{p}_i)^{e_i} = \prod_ip^{f_ie_i}$.
\end{proof}

\begin{theorem}[Kummer--Dedekind]\label{algebraic-number-theory:theorem-kummer-dedekind}
Suppose $\mathcal{O}_K = \ZZ[\theta]$, let $f$ be the minimal polynomial of $\theta$, and let $\bar f = \prod_{i=1}^g\bar g_i^{e_i}$ be the factorisation of its reduction modulo $p$ into distinct monic
irreducibles $\bar g_i \in \mathbb{F}_p[x]$, with monic lifts $g_i \in \ZZ[x]$. Then $\mathfrak{p}_i = (p, g_i(\theta))$ are distinct primes, $N(\mathfrak{p}_i) = p^{\deg g_i}$, and
\[
p\mathcal{O}_K = \mathfrak{p}_1^{e_1}\cdots\mathfrak{p}_g^{e_g}.
\]
\end{theorem}

\begin{proof}
\emph{Step 1: the primes above $p$.} $\mathcal{O}_K/p\mathcal{O}_K = \ZZ[x]/(f, p) \cong \mathbb{F}_p[x]/(\bar f)$. Its maximal ideals are the $(\bar g_i)/(\bar f)$, with quotients
$\mathbb{F}_p[x]/(\bar g_i)$ of order $p^{\deg g_i}$; their preimages in $\mathcal{O}_K$ are the $\mathfrak{p}_i = (p, g_i(\theta))$, and these are all the primes containing $p$.

\emph{Step 2: a containment.} $\prod\mathfrak{p}_i^{e_i} \subseteq (p, \prod g_i(\theta)^{e_i})$, and $\prod g_i^{e_i} \equiv f \pmod p$, so $\prod g_i(\theta)^{e_i} \in f(\theta) + p\ZZ[\theta] = p\ZZ[\theta]$.
Hence $\prod\mathfrak{p}_i^{e_i} \subseteq p\mathcal{O}_K$.

\emph{Step 3: equality.} Write $p\mathcal{O}_K = \prod\mathfrak{p}_i^{a_i}$. By Step 2 and Theorem~\ref{algebraic-number-theory:theorem-fractional-ideals}(2), $a_i \leq e_i$. By
Theorem~\ref{algebraic-number-theory:theorem-sum-ef}, $\sum a_i\deg g_i = n = \deg f = \sum e_i\deg g_i$, so $a_i = e_i$.
\end{proof}

The hypothesis $\mathcal{O}_K = \ZZ[\theta]$ can be weakened to $p \nmid [\mathcal{O}_K : \ZZ[\theta]]$, with the same proof applied to the localisation at $p$ \cite[Chapter I, Proposition 8.3]{neukirch-ant}.
It cannot be dropped altogether: there are number fields with $\mathcal{O}_K \neq \ZZ[\theta]$ for every $\theta$, the first example being a cubic field found by Dedekind \cite{neukirch-ant}.

\begin{corollary}\label{algebraic-number-theory:corollary-ramification-discriminant}
If $\mathcal{O}_K = \ZZ[\theta]$, then $p$ ramifies in $K$ if and only if $p \mid d_K$.
\end{corollary}

\begin{proof}
By the theorem, $p$ ramifies if and only if $\bar f$ has a repeated factor, if and only if $\bar f$ has a repeated root in $\bar{\mathbb{F}}_p$, if and only if the discriminant
$\prod_{i<j}(\bar\theta_i - \bar\theta_j)^2$ of $\bar f$ vanishes. This discriminant is a polynomial with integer coefficients in the coefficients of $f$, so it is the reduction of
$\prod_{i<j}(\theta_i - \theta_j)^2 = d(1, \dots, \theta^{n-1}) = d_K$ (Proposition~\ref{algebraic-number-theory:proposition-discriminant-power-basis}).
\end{proof}

The statement holds for every number field (Dedekind's theorem): the ramified primes are exactly the divisors of $d_K$ \cite[Chapter III, Corollary 2.12]{neukirch-ant}. In particular only finitely many
primes ramify.

\begin{theorem}[Splitting in quadratic fields]\label{algebraic-number-theory:theorem-quadratic-splitting}
Let $K = \QQ(\sqrt d)$ with $d$ squarefree, and $p$ an odd prime. Then $p$ ramifies if $p \mid d$, splits if $(\frac dp) = 1$, and is inert if $(\frac dp) = -1$. The prime $2$ ramifies if
$d \equiv 2, 3 \pmod 4$, splits if $d \equiv 1 \pmod 8$, and is inert if $d \equiv 5 \pmod 8$.
\end{theorem}

\begin{proof}
Apply Theorem~\ref{algebraic-number-theory:theorem-kummer-dedekind} with Theorem~\ref{algebraic-number-theory:theorem-quadratic-integers}. If $d \equiv 2, 3 \pmod 4$, $f = x^2 - d$; if $d \equiv 1 \pmod 4$,
$f = x^2 - x - \frac{d - 1}{4}$, whose discriminant is $d$. For odd $p$ both reduce to polynomials of discriminant $d$ times a square unit, which factor into distinct linear factors, a square, or stay
irreducible according as $d$ is a nonzero square, zero, or a non-square modulo $p$. For $p = 2$: $x^2 - d \equiv x^2$ or $(x + 1)^2$; and $x^2 - x - \frac{d - 1}{4} \equiv x(x + 1)$ if $\frac{d - 1}{4}$ is even,
that is $d \equiv 1 \pmod 8$, and $\equiv x^2 + x + 1$, irreducible over $\mathbb{F}_2$, if $d \equiv 5 \pmod 8$.
\end{proof}

\begin{example}\label{algebraic-number-theory:example-quadratic-splitting}
In $\ZZ[i]$ ($d = -1$) an odd prime splits if and only if $(\frac{-1}{p}) = 1$, that is $p \equiv 1 \pmod 4$, and $2 = -i(1 + i)^2$ ramifies: Theorem~\ref{number-theory:theorem-gaussian-primes} again. In
$\QQ(\sqrt{-5})$: $2$, $5$ ramify; $3$ and $7$ split ($-5 \equiv 1$ and $\equiv 2 = 3^2$ modulo $3$ and $7$); $11$ is inert ($-5 \equiv 6$ is not a square modulo $11$).
\end{example}

\section{Galois extensions and Frobenius elements}\label{algebraic-number-theory:section-galois}

Let $K/\QQ$ be Galois with group $G$. Each $\sigma \in G$ maps $\mathcal{O}_K$ to itself, hence permutes the primes above $p$.

\begin{proposition}\label{algebraic-number-theory:proposition-transitive}
$G$ acts transitively on the primes above $p$. Consequently all $e_i$ are equal to some $e$, all $f_i$ to some $f$, and $efg = n$.
\end{proposition}

\begin{proof}
Let $\mathfrak{p}$, $\mathfrak{q}$ lie above $p$, and suppose $\mathfrak{q} \neq \sigma\mathfrak{p}$ for all $\sigma$. By the Chinese remainder theorem (Theorem~\ref{rings:theorem-chinese-remainder}) choose
$x \in \mathcal{O}_K$ with $x \equiv 0 \pmod{\mathfrak{q}}$ and $x \equiv 1 \pmod{\sigma\mathfrak{p}}$ for all $\sigma$. Then $N(x) = \prod_\sigma\sigma(x) \in \mathfrak{q} \cap \ZZ = p\ZZ \subseteq \mathfrak{p}$, so some
$\sigma(x) \in \mathfrak{p}$, that is, $x \in \sigma^{-1}\mathfrak{p}$, a contradiction. As $\sigma(p\mathcal{O}_K) = p\mathcal{O}_K$, $\sigma$ preserves exponents, and it induces isomorphisms of residue fields; then
Theorem~\ref{algebraic-number-theory:theorem-sum-ef} gives $gef = n$.
\end{proof}

\begin{definition}\label{algebraic-number-theory:definition-decomposition-group}
For $\mathfrak{p}$ above $p$, the \emph{decomposition group} is $D_\mathfrak{p} = \{\sigma \in G : \sigma\mathfrak{p} = \mathfrak{p}\}$; each $\sigma \in D_\mathfrak{p}$ induces an automorphism $\bar\sigma$ of
$k_\mathfrak{p} = \mathcal{O}_K/\mathfrak{p}$, and the \emph{inertia group} $I_\mathfrak{p}$ is the kernel of $D_\mathfrak{p} \to \mathrm{Gal}(k_\mathfrak{p}/\mathbb{F}_p)$.
\end{definition}

\begin{theorem}\label{algebraic-number-theory:theorem-frobenius}
$|D_\mathfrak{p}| = ef$, and $D_\mathfrak{p} \to \mathrm{Gal}(k_\mathfrak{p}/\mathbb{F}_p)$ is surjective, so $|I_\mathfrak{p}| = e$. If $p$ is unramified, there is a unique $\mathrm{Frob}_\mathfrak{p} \in G$, the
\emph{Frobenius element}, with
\[
\mathrm{Frob}_\mathfrak{p}(x) \equiv x^p \pmod{\mathfrak{p}} \quad \text{for all } x \in \mathcal{O}_K;
\]
it generates $D_\mathfrak{p}$, has order $f$, and $\mathrm{Frob}_{\tau\mathfrak{p}} = \tau\mathrm{Frob}_\mathfrak{p}\tau^{-1}$. In particular an unramified $p$ splits completely if and only if $\mathrm{Frob}_\mathfrak{p} = 1$.
\end{theorem}

\begin{proof}
\emph{Order.} By transitivity and the orbit-stabiliser theorem, $|D_\mathfrak{p}| = n/g = ef$.

\emph{Surjectivity.} The group $k_\mathfrak{p}^\times$ is cyclic (Lemma~\ref{fields:lemma-cyclic}); choose $\theta \in \mathcal{O}_K$ whose image generates it and with $\theta \in \sigma\mathfrak{p}$ for every
$\sigma\mathfrak{p} \neq \mathfrak{p}$ (Chinese remainder theorem). The polynomial $h(x) = \prod_{\sigma \in G}(x - \sigma\theta)$ has integer coefficients and $\bar h(\bar\theta) = 0$, so $\bar h(\bar\theta^p) = \bar h(\bar\theta)^p = 0$
and $\bar\theta^p = \overline{\sigma\theta}$ for some $\sigma$. If $\sigma \notin D_\mathfrak{p}$, then $\sigma^{-1}\mathfrak{p} \neq \mathfrak{p}$, so $\theta \in \sigma^{-1}\mathfrak{p}$ and $\overline{\sigma\theta} = 0 \neq \bar\theta^p$. So
$\sigma \in D_\mathfrak{p}$ and $\bar\sigma$ agrees with the Frobenius $y \mapsto y^p$ on the generator $\bar\theta$, hence everywhere. The Frobenius generates $\mathrm{Gal}(k_\mathfrak{p}/\mathbb{F}_p)$, of order $f$
(Theorem~\ref{fields:theorem-finite-fields}), so the map is onto and $|I_\mathfrak{p}| = ef/f = e$.

\emph{Unramified case.} If $e = 1$, $D_\mathfrak{p} \to \mathrm{Gal}(k_\mathfrak{p}/\mathbb{F}_p)$ is an isomorphism, and $\mathrm{Frob}_\mathfrak{p}$ is the preimage of the Frobenius. Uniqueness in $G$: an element with the
congruence property maps $\mathfrak{p}$ into $\mathfrak{p}$ (if $x \in \mathfrak{p}$ then $\sigma x \equiv x^p \equiv 0$), so lies in $D_\mathfrak{p}$, where it is determined. Conjugation: $\tau\mathrm{Frob}_\mathfrak{p}\tau^{-1}$
has the congruence property modulo $\tau\mathfrak{p}$. Finally $p$ splits completely iff $g = n$ iff $f = 1$ iff $\mathrm{Frob}_\mathfrak{p} = 1$.
\end{proof}

\begin{remark}\label{algebraic-number-theory:remark-frobenius-restriction}
If $F \subseteq K$ is an intermediate field, Galois over $\QQ$, and $p$ is unramified in $K$, then the Frobenius of $\mathfrak{p} \cap \mathcal{O}_F$ in $\mathrm{Gal}(F/\QQ)$ is the restriction of
$\mathrm{Frob}_\mathfrak{p}$: the congruence $\mathrm{Frob}_\mathfrak{p}(x) \equiv x^p \pmod{\mathfrak{p}}$ holds in particular for $x \in \mathcal{O}_F$, modulo $\mathfrak{p} \cap \mathcal{O}_F$. Since conjugate Frobenius
elements are equal in an abelian group, for abelian $K$ one writes $\mathrm{Frob}_p$. Chebotarev's density theorem says that every conjugacy class of $G$ is the Frobenius class of a set of primes of
density $|C|/|G|$ \cite[Chapter VII, Theorem 13.4]{neukirch-ant}; it generalises Dirichlet's theorem on primes in progressions.
\end{remark}

\section{Class groups and units}\label{algebraic-number-theory:section-class-groups-and-units}

\subsection{The Minkowski embedding}\label{algebraic-number-theory:subsection-minkowski}

Let $\sigma_1, \dots, \sigma_{r_1}$ be the real embeddings of $K$, and let the remaining $2r_2$ embeddings be the pairs $\tau_j$, $\bar\tau_j$ ($j = 1, \dots, r_2$) of complex conjugate non-real ones, so
that $r_1 + 2r_2 = n$. The \emph{Minkowski embedding} is
\[
j \colon K \to \RR^{r_1} \times \CC^{r_2} \cong \RR^n, \qquad x \mapsto \bigl(\sigma_1(x), \dots, \sigma_{r_1}(x), \tau_1(x), \dots, \tau_{r_2}(x)\bigr),
\]
where $\CC$ is identified with $\RR^2$ by real and imaginary parts.

\begin{proposition}\label{algebraic-number-theory:proposition-minkowski-lattice}
For a nonzero ideal $\mathfrak{a}$, $j(\mathfrak{a})$ is a lattice in $\RR^n$ of covolume $2^{-r_2}\sqrt{|d_K|}\,N(\mathfrak{a})$.
\end{proposition}

\begin{proof}
Let $\omega_1, \dots, \omega_n$ be an integral basis. The matrix with columns $j(\omega_k)$ is obtained from $S = (\sigma_i(\omega_k))$, with rows indexed by all $n$ embeddings, by replacing each pair of
rows $\tau_j$, $\bar\tau_j$ by $\operatorname{Re}\tau_j = \frac12(\tau_j + \bar\tau_j)$ and $\operatorname{Im}\tau_j = \frac{1}{2i}(\tau_j - \bar\tau_j)$; this row operation multiplies the determinant by a number of
absolute value $\frac12$ for each pair. So the covolume of $j(\mathcal{O}_K)$ is $2^{-r_2}|\det S| = 2^{-r_2}\sqrt{|d_K|}$, which is nonzero; $j(\mathfrak{a})$ has index $N(\mathfrak{a})$ in it
(Proposition~\ref{geometry-of-numbers:proposition-index}).
\end{proof}

\begin{lemma}\label{algebraic-number-theory:lemma-volume-octahedron}
For $t > 0$, the set $B_t = \{(x, z) \in \RR^{r_1} \times \CC^{r_2} : \sum_i|x_i| + 2\sum_j|z_j| \leq t\}$ is compact, convex and symmetric, of volume $2^{r_1}(\frac\pi2)^{r_2}\frac{t^n}{n!}$.
\end{lemma}

\begin{proof}
Compactness, convexity and symmetry are clear. By homogeneity the volume is $c(r_1, r_2)t^n$, with $c(0, 0) = 1$. By Fubini's theorem in a real coordinate,
$c(r_1, r_2)t^n = \int_{-t}^{t}c(r_1 - 1, r_2)(t - |x|)^{n-1}\,dx = \frac{2}{n}c(r_1 - 1, r_2)t^n$. In a complex coordinate, using polar coordinates,
$c(r_1, r_2)t^n = \int_0^{t/2}2\pi\rho\,c(r_1, r_2 - 1)(t - 2\rho)^{n-2}\,d\rho = \frac\pi2c(r_1, r_2 - 1)\int_0^tu(t - u)^{n-2}\,du = \frac{\pi}{2}\frac{c(r_1, r_2 - 1)}{n(n - 1)}t^n$. Induction gives the formula.
\end{proof}

\subsection{Finiteness of the class number}\label{algebraic-number-theory:subsection-class-number}

\begin{theorem}[Minkowski bound]\label{algebraic-number-theory:theorem-minkowski-bound}
Every ideal class of $K$ contains an ideal $\mathfrak{b} \subseteq \mathcal{O}_K$ with
\[
N(\mathfrak{b}) \leq M_K = \Bigl(\frac4\pi\Bigr)^{r_2}\frac{n!}{n^n}\sqrt{|d_K|}.
\]
Consequently the class group is finite, and it is generated by the classes of the primes of norm at most $M_K$.
\end{theorem}

\begin{proof}
\emph{Step 1: a small element.} Let $\mathfrak{a}$ be a nonzero ideal, and choose $t$ with $t^n = n!(\frac4\pi)^{r_2}\sqrt{|d_K|}N(\mathfrak{a})$. Then
$\operatorname{vol}(B_t) = 2^n \cdot 2^{-r_2}\sqrt{|d_K|}N(\mathfrak{a})$ (Lemma~\ref{algebraic-number-theory:lemma-volume-octahedron}, using $n - r_1 = 2r_2$), so by Minkowski's theorem
(Theorem~\ref{geometry-of-numbers:theorem-minkowski}, compact case) and Proposition~\ref{algebraic-number-theory:proposition-minkowski-lattice} there is $0 \neq \alpha \in \mathfrak{a}$ with $j(\alpha) \in B_t$. The
quantity $\sum_i|x_i| + 2\sum_j|z_j|$ at $j(\alpha)$ is $\sum_{k=1}^n|\sigma_k(\alpha)|$ over all embeddings, so by the inequality of arithmetic and geometric means
\[
|N(\alpha)| = \prod_{k=1}^n|\sigma_k(\alpha)| \leq \Bigl(\frac tn\Bigr)^n = M_KN(\mathfrak{a}).
\]

\emph{Step 2: the class.} Given a class $c$, let $\mathfrak{a}$ be an ideal in $c^{-1}$ (multiply any fractional ideal of $c^{-1}$ by a suitable integer). With $\alpha$ as in Step 1,
$\mathfrak{b} = \alpha\mathfrak{a}^{-1}$ is an ideal (as $\alpha \in \mathfrak{a}$) in the class $c$, and $N(\mathfrak{b}) = |N(\alpha)|/N(\mathfrak{a}) \leq M_K$.

\emph{Step 3: finiteness.} There are finitely many ideals of norm at most $M_K$ (Proposition~\ref{algebraic-number-theory:proposition-ideal-norm}(3)), and each is a product of primes of norm at most $M_K$.
\end{proof}

\begin{corollary}[Minkowski]\label{algebraic-number-theory:corollary-discriminant-bound}
If $K \neq \QQ$, then $|d_K| > 1$; so every number field other than $\QQ$ has a ramified prime.
\end{corollary}

\begin{proof}
An ideal has norm at least $1$, so $M_K \geq 1$, that is, $\sqrt{|d_K|} \geq (\frac\pi4)^{r_2}\frac{n^n}{n!} \geq (\frac\pi4)^{n/2}\frac{n^n}{n!}$. The right side is $\frac{\pi}{2} > 1$ for $n = 2$ and increases
with $n$ (the ratio of consecutive values is $\sqrt{\pi/4}\,(1 + \frac1n)^n \geq \sqrt{\pi/4} \cdot 2 > 1$). The second statement follows from Dedekind's theorem quoted after
Corollary~\ref{algebraic-number-theory:corollary-ramification-discriminant}.
\end{proof}

\begin{example}\label{algebraic-number-theory:example-class-numbers}
\begin{enumerate}
\item $\QQ(i)$: $M_K = \frac4\pi \cdot \frac24 \cdot 2 = \frac4\pi < 2$, so $h = 1$; likewise $\QQ(\sqrt{-3})$ ($M_K \approx 1.10$), $\QQ(\sqrt 2)$ ($M_K = \frac12\sqrt8 \approx 1.41$), $\QQ(\sqrt 5)$ ($M_K \approx 1.12$).
\item $\QQ(\sqrt{-5})$: $M_K = \frac2\pi\sqrt{20} \approx 2.85$. The class group is generated by the primes of norm $2$, that is, by $\mathfrak{p} = (2, 1 + \sqrt{-5})$ with $\mathfrak{p}^2 = (2)$, which is not principal
(Example~\ref{algebraic-number-theory:example-sqrt-minus-five}). So $\mathrm{Cl}_K \cong \ZZ/2$ and $h = 2$.
\item The imaginary quadratic fields with $h = 1$ are $\QQ(\sqrt{-d})$ for $d = 1, 2, 3, 7, 11, 19, 43, 67, 163$ (Heegner, Baker, Stark); this is the list of Heegner numbers of
Chapter~\ref{number-theory}; that $n^2 + n + 41$ is prime for $0 \leq n \leq 39$ reflects $h(\QQ(\sqrt{-163})) = 1$ (Rabinowitsch's criterion).
\end{enumerate}
\end{example}

\subsection{Dirichlet's unit theorem}\label{algebraic-number-theory:subsection-units}

\begin{theorem}[Dirichlet]\label{algebraic-number-theory:theorem-units}
Let $\mu_K$ be the group of roots of unity in $K$, which is finite and cyclic. Then
\[
\mathcal{O}_K^\times \cong \mu_K \times \ZZ^{r_1 + r_2 - 1}.
\]
\end{theorem}

\begin{proof}
Put $r = r_1 + r_2$ and, for the embeddings $\rho_1, \dots, \rho_r$ consisting of the $\sigma_i$ and the $\tau_j$, let $n_k = 1$ for real and $n_k = 2$ for complex $\rho_k$.

\emph{Step 1: the logarithmic embedding.} $\ell(u) = (n_1\log|\rho_1(u)|, \dots, n_r\log|\rho_r(u)|)$ is a homomorphism $\mathcal{O}_K^\times \to \RR^r$ whose image lies in the hyperplane
$H = \{\sum y_k = 0\}$, since $\sum n_k\log|\rho_k(u)| = \log|N(u)| = 0$.

\emph{Step 2: kernel and discreteness.} For $C > 0$, the units with $|\rho_k(u)| \leq C$ for all $k$ are finite in number, because $j(\mathcal{O}_K)$ is a lattice and they lie in a bounded set. Hence
$\ker\ell$ is a finite group, so its elements are roots of unity; conversely roots of unity have $|\rho_k(\zeta)| = 1$. So $\ker\ell = \mu_K$, finite, and cyclic as a finite subgroup of $\CC^\times$. Also $\ell(\mathcal{O}_K^\times)$
meets every bounded set in finitely many points, so it is a discrete subgroup of $H$, hence a lattice in the subspace it spans (Theorem~\ref{geometry-of-numbers:theorem-discrete}), free of rank at most $r - 1$.

\emph{Step 3: elements of bounded norm with prescribed sizes.} Let $C_K = (\frac2\pi)^{r_2}\sqrt{|d_K|}$. For positive numbers $c_1, \dots, c_r$ with $\prod c_k^{n_k} = C_K$, the box
$\{|x_i| \leq c_i, |z_j| \leq c_{r_1 + j}\}$ has volume $2^{r_1}\pi^{r_2}\prod c_k^{n_k} = 2^n \cdot 2^{-r_2}\sqrt{|d_K|}$, so it contains $j(\alpha)$ for some $0 \neq \alpha \in \mathcal{O}_K$, and then
$|N(\alpha)| \leq C_K$.

\emph{Step 4: a unit small at all places but one.} Fix $k$. Starting from any $\alpha_1 \neq 0$, use Step 3 to choose $\alpha_{m+1} \neq 0$ with $|\rho_i(\alpha_{m+1})| < |\rho_i(\alpha_m)|$ for all $i \neq k$
(take $c_i < |\rho_i(\alpha_m)|$ for $i \neq k$ and $c_k$ determined by the product). All $\alpha_m\mathcal{O}_K$ have norm at most $C_K$, so two coincide, $\alpha_m\mathcal{O}_K = \alpha_{m^{\prime}}\mathcal{O}_K$ with
$m < m^{\prime}$, and $u_k = \alpha_{m^{\prime}}/\alpha_m$ is a unit with $\ell_i(u_k) < 0$ for all $i \neq k$, hence $\ell_k(u_k) > 0$.

\emph{Step 5: full rank.} Let $A$ be the $(r - 1) \times (r - 1)$ matrix $(\ell_i(u_k))_{i, k < r}$. Its off-diagonal entries are negative, and each column sum is $-\ell_r(u_k) > 0$. Such a matrix is invertible:
if $A^Tx = 0$ with $x \neq 0$, choose $k$ with $|x_k|$ maximal, say $x_k > 0$; then $0 = \sum_i\ell_i(u_k)x_i \geq x_k\sum_{i < r}\ell_i(u_k) > 0$, a contradiction. So $\ell(u_1), \dots, \ell(u_{r-1})$ are
linearly independent, and $\ell(\mathcal{O}_K^\times)$ has rank $r - 1$.

\emph{Step 6: conclusion.} The exact sequence $1 \to \mu_K \to \mathcal{O}_K^\times \to \ell(\mathcal{O}_K^\times) \cong \ZZ^{r-1} \to 0$ splits because the quotient is free.
\end{proof}

A basis $\varepsilon_1, \dots, \varepsilon_{r-1}$ of $\mathcal{O}_K^\times/\mu_K$ is a system of \emph{fundamental units}, and the absolute value of any $(r - 1) \times (r - 1)$ minor of the matrix
$(\ell_i(\varepsilon_k))$ is the \emph{regulator} $R_K$; it is the covolume of $\ell(\mathcal{O}_K^\times)$ suitably normalised, and does not depend on the choices.

\begin{example}\label{algebraic-number-theory:example-units}
\begin{enumerate}
\item Imaginary quadratic fields have $r = 1$: the units are the roots of unity, $\{\pm1\}$ except $\{\pm1, \pm i\}$ for $\QQ(i)$ and the sixth roots of unity for $\QQ(\sqrt{-3})$.
\item Real quadratic fields have $r = 2$: $\mathcal{O}_K^\times = \{\pm\varepsilon^k\}$. For $d \equiv 2, 3 \pmod 4$ the units $x + y\sqrt d$ are the solutions of $x^2 - dy^2 = \pm1$, found by the continued
fraction of $\sqrt d$ (Theorem~\ref{continued-fractions:theorem-pell}): $\varepsilon = 1 + \sqrt2$ for $d = 2$, $2 + \sqrt3$ for $d = 3$. For $\QQ(\sqrt5)$, $\varepsilon = \varphi = \frac{1 + \sqrt5}{2}$, of
norm $-1$, and $\varphi^k = \frac{L_k + F_k\sqrt5}{2}$ with the Lucas and Fibonacci numbers.
\item $K = \QQ(\sqrt[3]{2})$ has $r_1 = 1$, $r_2 = 1$, so the unit group has rank $1$. The element $\sqrt[3]{2} - 1$ is a unit, with inverse $\sqrt[3]{4} + \sqrt[3]{2} + 1$, and it has infinite order since
its real absolute value $0.26\ldots$ is not $1$.
\end{enumerate}
\end{example}

\begin{remark}\label{algebraic-number-theory:remark-class-number-formula}
The \emph{Dedekind zeta function} $\zeta_K(s) = \sum_{\mathfrak{a}}N(\mathfrak{a})^{-s} = \prod_{\mathfrak{p}}(1 - N(\mathfrak{p})^{-s})^{-1}$ (sum over nonzero ideals, product over nonzero primes) converges for
$\operatorname{Re}s > 1$; the Euler product expresses unique factorisation of ideals. It extends meromorphically to $\CC$ with a simple pole at $s = 1$, whose residue is given by the \emph{analytic class
number formula}
\[
\lim_{s \to 1}(s - 1)\zeta_K(s) = \frac{2^{r_1}(2\pi)^{r_2}h_KR_K}{|\mu_K|\sqrt{|d_K|}}
\]
\cite[Chapter VII, Corollary 5.11]{neukirch-ant}. For imaginary quadratic $K$ it reduces to Dirichlet's formula (Remark~\ref{analytic-number-theory:remark-class-number-formula}), since
$\zeta_K(s) = \zeta(s)L(s, \chi_{d_K})$. The proof counts ideals in a class by counting lattice points of $j(\mathfrak{a})$ in a cone, using the two finiteness theorems above.
\end{remark}

\section{Cyclotomic fields}\label{algebraic-number-theory:section-cyclotomic-fields}

Let $p$ be an odd prime, $\zeta = e^{2\pi i/p}$ and $K = \QQ(\zeta)$, of degree $p - 1$ with Galois group $(\ZZ/p)^\times$, $a \mapsto (\sigma_a \colon \zeta \mapsto \zeta^a)$
(Theorem~\ref{fields:theorem-cyclotomic}).

\begin{theorem}\label{algebraic-number-theory:theorem-cyclotomic-integers}
\begin{enumerate}
\item $\mathcal{O}_K = \ZZ[\zeta]$, and $d_K = (-1)^{(p-1)/2}p^{p-2}$.
\item $p$ is totally ramified: $p\mathcal{O}_K = (1 - \zeta)^{p-1}$, with $(1 - \zeta)$ prime of norm $p$.
\item For a prime $q \neq p$, let $f$ be the order of $q$ in $(\ZZ/p)^\times$. Then $q\mathcal{O}_K$ is a product of $(p - 1)/f$ distinct primes of residue degree $f$, and $\mathrm{Frob}_q = \sigma_q$. In
particular $q$ splits completely if and only if $q \equiv 1 \pmod p$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) \emph{Discriminant of $\ZZ[\zeta]$.} From $(x - 1)\Phi_p(x) = x^p - 1$ we get $(\zeta - 1)\Phi_p^{\prime}(\zeta) = p\zeta^{p-1}$. Now $N(\zeta) = \pm1$ (a unit) and $N(1 - \zeta) = \prod_a(1 - \zeta^a) = \Phi_p(1) = p$,
so $|N(\Phi_p^{\prime}(\zeta))| = p^{p-1}/p = p^{p-2}$, and by Proposition~\ref{algebraic-number-theory:proposition-discriminant-power-basis} $d(1, \zeta, \dots, \zeta^{p-2}) = \pm p^{p-2}$. All embeddings of
$K$ are non-real, so $r_2 = \frac{p-1}{2}$ and the sign is $(-1)^{(p-1)/2}$ (remark after Definition~\ref{algebraic-number-theory:definition-discriminant}). \emph{Index.} By Theorem~\ref{algebraic-number-theory:theorem-integral-basis}, $[\mathcal{O}_K : \ZZ[\zeta]]^2$ divides $p^{p-2}$, so the index is a power of $p$. The minimal polynomial of
$\zeta - 1$ is $\Phi_p(x + 1) = \frac{(x + 1)^p - 1}{x} = x^{p-1} + \binom p1x^{p-2} + \dots + \binom{p}{p-1}$, which is Eisenstein at $p$; by Lemma~\ref{algebraic-number-theory:lemma-eisenstein-index},
$p \nmid [\mathcal{O}_K : \ZZ[\zeta - 1]] = [\mathcal{O}_K : \ZZ[\zeta]]$. So the index is $1$.

(2) $\Phi_p(x + 1) \equiv x^{p-1} \pmod p$, so by Theorem~\ref{algebraic-number-theory:theorem-kummer-dedekind} applied to $\theta = \zeta - 1$, $p\mathcal{O}_K = (p, \zeta - 1)^{p-1}$, and
$(p, \zeta - 1) = (1 - \zeta)$ since $1 - \zeta$ divides $p = \prod_a(1 - \zeta^a)$.

(3) $q$ is unramified, as $q \nmid d_K$ (Corollary~\ref{algebraic-number-theory:corollary-ramification-discriminant}). The automorphism $\sigma_q$ satisfies
$\sigma_q(\sum a_i\zeta^i) = \sum a_i\zeta^{qi} \equiv (\sum a_i\zeta^i)^q \pmod q$ (the Frobenius in characteristic $q$ and Fermat's little theorem for the $a_i$), so $\sigma_q = \mathrm{Frob}_\mathfrak{q}$ for every
$\mathfrak{q}$ above $q$, by the uniqueness in Theorem~\ref{algebraic-number-theory:theorem-frobenius}. Its order is the order $f$ of $q$ modulo $p$, which is the residue degree, and $g = (p - 1)/f$.
\end{proof}

\begin{proposition}[Gauss sums]\label{algebraic-number-theory:proposition-gauss-sum}
Let $g = \sum_{a=1}^{p-1}\bigl(\frac ap\bigr)\zeta^a \in \ZZ[\zeta]$. Then $g^2 = p^*$, where $p^* = (-1)^{(p-1)/2}p$. Hence the unique quadratic subfield of $\QQ(\zeta)$ is $\QQ(\sqrt{p^*})$.
\end{proposition}

\begin{proof}
\emph{Computation.} $g^2 = \sum_{a,b}(\frac{ab}{p})\zeta^{a+b}$. Substituting $b = ac$ for fixed $a$ ($c$ runs through $(\ZZ/p)^\times$ as $b$ does) and using $(\frac{a^2c}{p}) = (\frac cp)$,
\[
g^2 = \sum_c\Bigl(\frac cp\Bigr)\sum_a\zeta^{a(1 + c)}.
\]
The inner sum is $p - 1$ for $c = -1$ and $-1$ otherwise (a sum of all nontrivial $p$-th roots of unity). Since $\sum_c(\frac cp) = 0$, $g^2 = (\frac{-1}{p})(p - 1) - \sum_{c \neq -1}(\frac cp) = (\frac{-1}{p})p$,
and $(\frac{-1}{p}) = (-1)^{(p-1)/2}$ by Euler's criterion (Proposition~\ref{number-theory:proposition-euler-criterion}).

\emph{The subfield.} $\mathrm{Gal}(K/\QQ) \cong (\ZZ/p)^\times$ is cyclic of even order, so it has a unique subgroup of index $2$ (the squares), hence a unique quadratic subfield; it contains $g$, which is
irrational since $g^2 = p^*$ is not a rational square.
\end{proof}

\begin{theorem}[Quadratic reciprocity, second proof]\label{algebraic-number-theory:theorem-reciprocity-cyclotomic}
For distinct odd primes $p$, $q$, $\bigl(\frac{p^*}{q}\bigr) = \bigl(\frac qp\bigr)$; equivalently $(\frac pq)(\frac qp) = (-1)^{\frac{p-1}2\frac{q-1}2}$.
\end{theorem}

\begin{proof}
Let $F = \QQ(\sqrt{p^*}) \subseteq K$. Since $p^* \equiv 1 \pmod 4$, Theorem~\ref{algebraic-number-theory:theorem-quadratic-splitting} says that $q$ splits in $F$ if and only if $(\frac{p^*}{q}) = 1$, that is, by
Theorem~\ref{algebraic-number-theory:theorem-frobenius}, if and only if the Frobenius of $q$ in $\mathrm{Gal}(F/\QQ)$ is trivial. By Remark~\ref{algebraic-number-theory:remark-frobenius-restriction} and
Theorem~\ref{algebraic-number-theory:theorem-cyclotomic-integers}(3), this Frobenius is the restriction of $\sigma_q$ to $F$, which is trivial if and only if $\sigma_q$ lies in the index-$2$ subgroup fixing $F$,
the squares, that is, if and only if $(\frac qp) = 1$. The equivalent form follows from $(\frac{p^*}{q}) = (\frac{-1}{q})^{(p-1)/2}(\frac pq)$ and $(\frac{-1}{q}) = (-1)^{(q-1)/2}$.
\end{proof}

The first proof, by counting lattice points, is Theorem~\ref{number-theory:theorem-quadratic-reciprocity}. The present one explains the law: the splitting of $q$ in the quadratic field $F$ is governed by a
congruence condition on $q$ because $F$ sits inside a cyclotomic field.

\begin{theorem}[Kronecker--Weber]\label{algebraic-number-theory:theorem-kronecker-weber}
Every finite abelian extension of $\QQ$ is contained in a cyclotomic field $\QQ(e^{2\pi i/m})$ \cite{neukirch-ant}.
\end{theorem}

\begin{remark}\label{algebraic-number-theory:remark-kummer-regular}
Kummer called an odd prime $p$ \emph{regular} if $p \nmid h_{\QQ(\zeta_p)}$, and proved that Fermat's equation $x^p + y^p = z^p$ then has no solution in positive integers; he also showed that $p$ is regular if and
only if $p$ divides none of the numerators of the Bernoulli numbers $B_2, B_4, \dots, B_{p-3}$ \cite[Theorem 5.34 and Chapter 9]{washington-cyclotomic}. The irregular primes below $100$ are $37$, $59$, $67$.
The idea is the one of Section~\ref{diophantine-equations:subsection-three}: factor $x^p + y^p = \prod(x + \zeta^ky)$ and use that coprime ideals whose product is a $p$-th power are $p$-th powers; regularity
lets one pass from ideals to elements, and Kummer's lemma on units does the rest.
\end{remark}

\section{Exercises}\label{algebraic-number-theory:section-exercises}

\begin{exercise}\label{algebraic-number-theory:exercise-cube-root-two}
Show that the ring of integers of $\QQ(\sqrt[3]{2})$ is $\ZZ[\sqrt[3]{2}]$ and that its discriminant is $-108$.
\end{exercise}

\begin{solution}
Let $\theta = \sqrt[3]{2}$, $f = x^3 - 2$. By Proposition~\ref{algebraic-number-theory:proposition-discriminant-power-basis}, $d(1, \theta, \theta^2) = (-1)^3N(3\theta^2) = -27N(\theta)^2 = -27 \cdot 4 = -108$. The
index $m = [\mathcal{O}_K : \ZZ[\theta]]$ satisfies $m^2 \mid 108 = 2^2 \cdot 3^3$, so $m \mid 6$. $f$ is Eisenstein at $2$, and $f(x - 1) = x^3 - 3x^2 + 3x - 3$ is Eisenstein at $3$, with $\ZZ[\theta - 1] = \ZZ[\theta]$.
By Lemma~\ref{algebraic-number-theory:lemma-eisenstein-index}, $2 \nmid m$ and $3 \nmid m$, so $m = 1$.
\end{solution}

\begin{exercise}\label{algebraic-number-theory:exercise-class-number-one}
Show that $\QQ(\sqrt{-2})$, $\QQ(\sqrt{-7})$ and $\QQ(\sqrt{-11})$ have class number $1$.
\end{exercise}

\begin{solution}
For imaginary quadratic fields $M_K = \frac2\pi\sqrt{|d_K|}$. For $d_K = -8, -7, -11$: $M_K \approx 1.80, 1.68, 2.11$. In the first two cases $M_K < 2$, so $h = 1$. For $\QQ(\sqrt{-11})$ we must look at the primes
of norm $2$: since $-11 \equiv 5 \pmod 8$, $2$ is inert (Theorem~\ref{algebraic-number-theory:theorem-quadratic-splitting}), so there are none, and $h = 1$.
\end{solution}

\begin{exercise}\label{algebraic-number-theory:exercise-sqrt-minus-six}
Show that $\QQ(\sqrt{-6})$ has class group $\ZZ/2$.
\end{exercise}

\begin{solution}
$d_K = -24$ and $M_K = \frac2\pi\sqrt{24} \approx 3.12$. The primes $2$ and $3$ ramify: $(2) = \mathfrak{p}_2^2$ with $\mathfrak{p}_2 = (2, \sqrt{-6})$, and $(3) = \mathfrak{p}_3^2$ with $\mathfrak{p}_3 = (3, \sqrt{-6})$. Neither is
principal, since $a^2 + 6b^2$ takes neither the value $2$ nor $3$. But $(\sqrt{-6})^2 = (6) = \mathfrak{p}_2^2\mathfrak{p}_3^2$, so $(\sqrt{-6}) = \mathfrak{p}_2\mathfrak{p}_3$ and $\mathfrak{p}_3$ is in the class of
$\mathfrak{p}_2^{-1} = \mathfrak{p}_2$. The class group is generated by the class of $\mathfrak{p}_2$, of order $2$.
\end{solution}

\begin{exercise}\label{algebraic-number-theory:exercise-split-cyclotomic}
Let $p$ be an odd prime. Using Theorem~\ref{algebraic-number-theory:theorem-cyclotomic-integers}, show that there are infinitely many primes $q \equiv 1 \pmod p$.
\end{exercise}

\begin{solution}
Suppose $q_1, \dots, q_k$ are all the primes $\equiv 1 \pmod p$, and let
$N = \Phi_p(pq_1\cdots q_k)$. Then $N \equiv 1 \pmod{pq_1 \cdots q_k}$ and $N > 1$, so it has a prime factor $q \neq p, q_i$. Modulo $q$, $\Phi_p$ has the root $x = pq_1\cdots q_k$, so $\bar\Phi_p$ has a
linear factor and by Theorem~\ref{algebraic-number-theory:theorem-kummer-dedekind} some prime above $q$ has residue degree $1$; then $f = 1$, so $q \equiv 1 \pmod p$, a contradiction.
\end{solution}

\begin{exercise}\label{algebraic-number-theory:exercise-units-norm}
Show that the units of $\ZZ[\sqrt3]$ all have norm $+1$, and deduce that $x^2 - 3y^2 = -1$ has no integer solution.
\end{exercise}

\begin{solution}
If $x^2 - 3y^2 = -1$, then $x^2 \equiv -1 \pmod 3$, impossible. So every unit $x + y\sqrt3$ has norm $+1$; the fundamental unit is $2 + \sqrt3$.
\end{solution}
